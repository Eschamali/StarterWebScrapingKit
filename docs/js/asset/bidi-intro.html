BiDi デモコード集：WebDriverBiDiCore の実力を見る

WebDriverBiDiCore.cls を使った実際の動くデモコードを紹介します。
ソースは VBAProject/Modules/Demo_WebDriverBiDi.bas に収録されています。

🌐 Demo 1：ネットワークイベントのキャプチャ

CDP版の「ネットワークイベント監視」を、BiDi の network ドメインで再現したデモです。
イベントの「セーブ＆ロード」機能（有効化・無効化・再開）を段階的に実演します。

💡
BiDiEvents プロパティの使い方（3パターン）
  • Set .BiDiEvents = New Dictionary → イベント記録を開始
  • Set .BiDiEvents = Nothing → イベントを破棄（無効化）
  • Set .BiDiEvents = 変数 → セーブデータから再開
Demoコードを見る（ネットワークイベントの確認） 
Sub ネットワークイベントの確認()
    Dim JsonDicObj As New WebJsonConverter
    Dim CharConvObj As New CharacterCodeConversion

    Dim Demo_NetworkEvent As New WebDriverBiDiCore
    Demo_NetworkEvent.start

    Dim paramsBiDi As Dictionary, resultBiDi As Dictionary

    ' コンテキストIDを取得
    Set resultBiDi = Demo_NetworkEvent.invokeMethod("browsingContext.getTree")
    Dim targetContext As String
    If Not (resultBiDi Is Nothing) Then
        targetContext = resultBiDi("contexts")(1)("context")
    End If

    ' ---- 機能1：イベントキャプチャ 有効化 ----
    Set Demo_NetworkEvent.BiDiEvents = New Dictionary    ' New Dictionary を渡す → 記録開始

    ' ネットワークイベントを購読
    Set paramsBiDi = New Dictionary
    Dim eventsArray As New Collection
    eventsArray.Add "network.beforeRequestSent"
    eventsArray.Add "network.responseCompleted"
    paramsBiDi.Add "events", eventsArray
    Demo_NetworkEvent.invokeMethod "session.subscribe", paramsBiDi

    ' ページ遷移して読み込み完了まで待機
    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "context", targetContext
    paramsBiDi.Add "url", "http://officetanaka.net/excel/vba/file/file11.htm"
    paramsBiDi.Add "wait", "complete"
    Demo_NetworkEvent.invokeMethod "browsingContext.navigate", paramsBiDi

    Demo_NetworkEvent.TakeEvents                         ' イベントをバッファから取り出す
    CharConvObj.BytesToSaveFile CharConvObj.BytesFromString(JsonDicObj.ConvertToJson(Demo_NetworkEvent.BiDiEvents)), _
        Environ("UserProfile") & "\Downloads", "BiDi_Event.json"

    ' ---- 機能2：セーブデータ作成 → キャプチャ 無効化 ----
    Dim SaveDataEvents As Dictionary: Set SaveDataEvents = Demo_NetworkEvent.BiDiEvents  ' セーブ
    Set Demo_NetworkEvent.BiDiEvents = Nothing           ' Nothing → 記録破棄

    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "context", targetContext
    paramsBiDi.Add "url", "http://officetanaka.net/youtube/20200714b.htm"
    paramsBiDi.Add "wait", "complete"
    Demo_NetworkEvent.invokeMethod "browsingContext.navigate", paramsBiDi
    Demo_NetworkEvent.TakeEvents
    ' ↓ 無効中なので BiDiEvents が Nothing のまま → 0バイトのファイルになる
    CharConvObj.BytesToSaveFile CharConvObj.BytesFromString(JsonDicObj.ConvertToJson(Demo_NetworkEvent.BiDiEvents)), _
        Environ("UserProfile") & "\Downloads", "BiDi_NotEvent.json"

    ' ---- 機能3：セーブデータから キャプチャ 再開 ----
    Set Demo_NetworkEvent.BiDiEvents = SaveDataEvents    ' セーブデータを復元 → 記録再開

    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "context", targetContext
    paramsBiDi.Add "url", "http://officetanaka.net/index.stm"
    paramsBiDi.Add "wait", "complete"
    Demo_NetworkEvent.invokeMethod "browsingContext.navigate", paramsBiDi
    Demo_NetworkEvent.TakeEvents
    CharConvObj.BytesToSaveFile CharConvObj.BytesFromString(JsonDicObj.ConvertToJson(Demo_NetworkEvent.BiDiEvents)), _
        Environ("UserProfile") & "\Downloads", "BiDi_EventFromSaveData.json"

    Demo_NetworkEvent.quit
End Sub

🧩 Demo 2：拡張機能のインストール ／ アンインストール

BiDiプロトコルの webExtension ドメインを使い、ローカルフォルダにある拡張機能を動的に読み込み、アンインストールするデモです。

⚠️
事前設定が必要

「ブラウザ起動設定」シートで 「CDP-Jsonで拡張機能を制御」をON にしてから実行してください。

Demoコードを見る（UseExtensions） 
Sub UseExtensions()
    Dim JsonDicObj As New WebJsonConverter

    ' 拡張機能フォルダをダイアログで選択
    Dim ExtensionsFolderPath As String
    With Application.FileDialog(msoFileDialogFolderPicker)
        .Title = "拡張機能の基となる`manifest.json`を含むフォルダを選択してください"
        .InitialFileName = Environ("UserProfile") & "\AppData\Local"
        If .show = -1 Then ExtensionsFolderPath = .SelectedItems(1) Else Exit Sub
    End With

    Dim controlExtensions As New WebDriverBiDiCore
    controlExtensions.start     ' ブラウザ起動

    Dim paramsBiDi As Dictionary, resultBiDi As Dictionary

    ' コンテキスト取得 & 拡張機能ページへ遷移
    Set resultBiDi = controlExtensions.invokeMethod("browsingContext.getTree")
    Dim targetContext As String
    If Not (resultBiDi Is Nothing) Then targetContext = resultBiDi("contexts")(1)("context")

    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "context", targetContext
    paramsBiDi.Add "url", "edge://extensions/"
    paramsBiDi.Add "wait", "complete"
    controlExtensions.invokeMethod "browsingContext.navigate", paramsBiDi

    ' ---- 拡張機能をインストール (webExtension.install) ----
    Set paramsBiDi = New Dictionary
    Dim extData As New Dictionary
    extData.Add "type", "path"
    extData.Add "path", ExtensionsFolderPath
    paramsBiDi.Add "context", targetContext
    paramsBiDi.Add "extensionData", extData
    Set resultBiDi = controlExtensions.invokeMethod("webExtension.install", paramsBiDi, StopError:=False)

    If resultBiDi Is Nothing Then
        MsgBox "インストール失敗：" & controlExtensions.LastBiDiJsonError("message"), vbCritical
        controlExtensions.quit: Exit Sub
    End If

    MsgBox "インストール成功！　OKでアンインストールします。", vbInformation, "ExtID：" & resultBiDi("extension")

    ' ---- 拡張機能をアンインストール (webExtension.uninstall) ----
    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "extension", resultBiDi("extension")
    Set resultBiDi = controlExtensions.invokeMethod("webExtension.uninstall", paramsBiDi, StopError:=False)

    If resultBiDi Is Nothing Then
        MsgBox "アンインストール失敗：" & controlExtensions.LastBiDiJsonError("message"), vbCritical
    Else
        MsgBox "アンインストール成功！", vbInformation
    End If

    controlExtensions.quit
End Sub

💬 Demo 3：JavaScriptダイアログの自動応答

alert / prompt ダイアログを BiDi の 非同期コマンド＋イベント監視で自動処理するデモです。
イベントが来るまでポーリングし、確認後に browsingContext.handleUserPrompt で応答します。

🔍
ポイント：非同期実行 + イベント待ち受けパターン

JS の click() を invokeMethodAsync で非同期発火 → TakeEvents ループで browsingContext.userPromptOpened が来るまで待機 → ダイアログへ応答、という流れです。

Demoコードを見る（TestAlert） 
Sub TestAlert()
    Dim Demo_alerts As New WebDriverBiDiCore

    ' unhandledPromptBehavior を "ignore" にしてダイアログを自動処理させない
    Dim caps As New Dictionary
    Dim alwaysMatch As New Dictionary
    alwaysMatch.Add "unhandledPromptBehavior", "ignore"
    caps.Add "capabilities", New Dictionary
    caps("capabilities").Add "alwaysMatch", alwaysMatch

    Demo_alerts.start "https://www.selenium.dev/selenium/web/alerts.html", , caps

    Dim paramsBiDi As Dictionary, resultBiDi As Dictionary
    Set resultBiDi = Demo_alerts.invokeMethod("browsingContext.getTree")
    Dim targetContext As String
    If Not (resultBiDi Is Nothing) Then targetContext = resultBiDi("contexts")(1)("context")

    Dim 入力文字内容 As String
    入力文字内容 = "VBAから入力したテスト文字列です！" & WorksheetFunction.Unichar(129418)

    With Demo_alerts
        ' userPromptOpened イベントを購読
        Set paramsBiDi = New Dictionary
        Dim eventsArray As New Collection
        eventsArray.Add "browsingContext.userPromptOpened"
        paramsBiDi.Add "events", eventsArray
        .invokeMethod "session.subscribe", paramsBiDi

        Dim i As Long
        For i = 1 To 3     ' alert / empty-alert / prompt の3種
            Dim targetID As String
            Select Case i
                Case 1: targetID = "alert"
                Case 2: targetID = "empty-alert"
                Case 3: targetID = "prompt"
            End Select

            ' イベントバッファをリセット
            Set .BiDiEvents = New Dictionary

            ' JS の click() を非同期で実行 → alert が発火する
            Set paramsBiDi = New Dictionary
            paramsBiDi.Add "expression", "document.getElementById('" & targetID & "').click()"
            Dim targetDict As New Dictionary
            targetDict.Add "context", targetContext
            paramsBiDi.Add "target", targetDict
            paramsBiDi.Add "awaitPromise", False
            Dim AsyncID As Long
            AsyncID = .invokeMethodAsync("script.evaluate", paramsBiDi)

            ' browsingContext.userPromptOpened が来るまでポーリング
            Const SearchEventName As String = "browsingContext.userPromptOpened"
            Do
                .TakeEvents
                If .BiDiEvents("EventMethods").Exists(SearchEventName) Then Exit Do
            Loop While True

            ' ダイアログへ応答
            Set paramsBiDi = New Dictionary
            paramsBiDi.Add "context", targetContext
            paramsBiDi.Add "accept", True
            paramsBiDi.Add "userText", 入力文字内容
            .invokeMethod "browsingContext.handleUserPrompt", paramsBiDi

            ' 非同期コマンドの結果も確認
            .sleep 0.5: .TakeEvents
            Dim エラー確認 As Boolean
            Dim resBiDiAsync As Dictionary
            Set resBiDiAsync = .ResultBiDiForAsync(AsyncID, エラー確認)
        Next i

        .quit
    End With
End Sub

🔌 Demo 4：BiDi+ — CDP コマンドのトンネリング

WebDriver BiDi にまだ存在しない機能は 「BiDi+」（Chromium独自拡張）を使うことで、 従来のCDPコマンドを goog:cdp.sendCommand 経由でトンネリング実行できます。

🚀
BiDi と CDP の「いいとこ取り」が可能

BiDi がカバーしていない領域も、同じ WebDriverBiDiCore インスタンスのまま CDP コマンドを呼び出せます。移行期における強力な逃げ道です。

Demoコードを見る（TestBiDiPlus_CDPTunnel） 
Sub TestBiDiPlus_CDPTunnel()
    Dim bidiPlus As New WebDriverBiDiCore
    bidiPlus.start

    Dim paramsBiDi As Dictionary, resultBiDi As Dictionary

    ' 1. BiDiコンテキストから CDP セッションIDを取得 (goog:cdp.getSession)
    Dim targetContext As String
    Set resultBiDi = bidiPlus.invokeMethod("browsingContext.getTree")
    If Not resultBiDi Is Nothing Then
        targetContext = resultBiDi("contexts")(1)("context")

        Set paramsBiDi = New Dictionary
        paramsBiDi.Add "context", targetContext
        Set resultBiDi = bidiPlus.invokeMethod("goog:cdp.getSession", paramsBiDi)

        Dim cdpSessionId As String
        If Not resultBiDi Is Nothing Then
            cdpSessionId = resultBiDi("session")
            MsgBox "CDPセッションIDを取得：" & cdpSessionId, vbInformation, "BiDi+ GetSession"
        End If
    End If

    ' 2. CDPコマンド「Browser.getVersion」を BiDi 経由で実行 (goog:cdp.sendCommand)
    Set paramsBiDi = New Dictionary
    paramsBiDi.Add "method", "Browser.getVersion"
    paramsBiDi.Add "params", New Dictionary
    If cdpSessionId <> "" Then paramsBiDi.Add "session", cdpSessionId

    Set resultBiDi = bidiPlus.invokeMethod("goog:cdp.sendCommand", paramsBiDi)

    If Not resultBiDi Is Nothing Then
        MsgBox "Browser.getVersion を BiDi 経由で実行成功！" & vbCrLf & _
               "UserAgent：" & resultBiDi("result")("userAgent") & vbCrLf & _
               "Protocol：" & resultBiDi("result")("protocolVersion"), vbInformation, "BiDi+ CDP Tunnel"
    End If

    bidiPlus.quit
End Sub
